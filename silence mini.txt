-- 1. Tìm model tủ duy nhất 1 lần (không lặp vô tận trên workspace)
local function FindCurrentLockerModel()
	if not hrp then return nil end
	local charPos = hrp.Position
	local closestLocker = nil
	local minDst = 15

	-- Tìm trong workspace
	for _, v in ipairs(workspace:GetChildren()) do
		if v:IsA("Model") or v:IsA("Folder") then
			for _, item in ipairs(v:GetChildren()) do
				if item:IsA("Model") and (item.Name:find("Locker") or item.Name:find("Hiding") or item.Name:find("Bed") or item.Name:find("Closet")) then
					local primary = item.PrimaryPart or item:FindFirstChildWhichIsA("BasePart")
					if primary then
						local dst = (primary.Position - charPos).Magnitude
						if dst < minDst then
							minDst = dst
							closestLocker = item
						end
					end
				end
			end
		end
	end
	return closestLocker
end

-- 2. Quét cache các script ngầm chống văng tủ duy nhất 1 lần
local function CacheAntiKickTargets()
	cachedAntiKickScripts = {}
	local searchContainers = {
		player:FindFirstChild("PlayerGui"),
		player:FindFirstChild("PlayerScripts"),
		character
	}

	for _, container in ipairs(searchContainers) do
		if container then
			for _, desc in ipairs(container:GetDescendants()) do
				local nameLower = desc.Name:lower()
				if nameLower:find("cleithro") or nameLower:find("claustro") or nameLower:find("locker") or nameLower:find("hiding") then
					table.insert(cachedAntiKickScripts, desc)
				end
			end
		end
	end
end

-- 3. Hàm áp dụng trạng thái Khóa / Mở Khóa tủ cực nhẹ
local function LockLockerSystem(isLocked)
	-- A. Báo Server qua RemoteEvent gốc của game
	pcall(function()
		local events = ReplicatedStorage:FindFirstChild("Events") or ReplicatedStorage:FindFirstChild("Remotes")
		if events then
			local doorLock = events:FindFirstChild("PandemoniumDoorLock") or events:FindFirstChild("LockLocker")
			if doorLock then
				doorLock:FireServer(isLocked and "lock" or "unlock")
			end
		end
	end)

	-- B. Khóa trực tiếp thuộc tính trên Model Tủ đã Cache
	if not cachedLocker or not cachedLocker.Parent then
		cachedLocker = FindCurrentLockerModel()
	end

	if cachedLocker then
		pcall(function()
			cachedLocker:SetAttribute("CantExit", isLocked or nil)
			cachedLocker:SetAttribute("Locked", isLocked or nil)
			cachedLocker:SetAttribute("InMinigame", isLocked or nil)
			cachedLocker:SetAttribute("InUse", isLocked or nil)
		end)
		for _, desc in ipairs(cachedLocker:GetChildren()) do
			if desc:IsA("BoolValue") and (desc.Name == "CantExit" or desc.Name == "Locked") then
				desc.Value = isLocked
			end
		end
	end

	-- C. Vô hiệu hóa các Script/Module đã được Cache sẵn (Không tốn CPU)
	for _, desc in ipairs(cachedAntiKickScripts) do
		if desc and desc.Parent then
			if isLocked then
				if desc:IsA("LocalScript") or desc:IsA("Script") then
					pcall(function() desc.Disabled = true end)
				end
				if getsenv and (desc:IsA("LocalScript") or desc:IsA("Script")) then
					pcall(function()
						local env = getsenv(desc)
						if env.Stop then env.Stop() end
						if env.Cancel then env.Cancel() end
						if env.CantExit ~= nil then env.CantExit = true end
					end)
				end
				if desc:IsA("ModuleScript") then
					pcall(function()
						local mod = require(desc)
						if type(mod) == "table" then
							if mod.Stop then pcall(mod.Stop) end
							mod.CantExit = true
						end
					end)
				end
			else
				if desc:IsA("LocalScript") or desc:IsA("Script") then
					pcall(function() desc.Disabled = false end)
				end
				if getsenv and (desc:IsA("LocalScript") or desc:IsA("Script")) then
					pcall(function()
						local env = getsenv(desc)
						if env.Start then env.Start(desc, 10, 15) end
						if env.CantExit ~= nil then env.CantExit = nil end
					end)
				end
				if desc:IsA("ModuleScript") then
					pcall(function()
						local mod = require(desc)
						if type(mod) == "table" then
							if mod.Start then pcall(mod.Start, desc, 10, 15) end
							mod.CantExit = nil
						end
					end)
				end
			end
		end
	end
end

-- 4. Hook chặn Remote Event văng tủ (Remote Interceptor)
pcall(function()
	if hookmetamethod and getrawmetatable then
		local oldNC
		oldNC = hookmetamethod(game, "__namecall", function(self, ...)
			local method = getnamecallmethod()
			if not Ended and (method == "FireServer" or method == "InvokeServer") then
				local name = tostring(self.Name):lower()
				if name:find("exit") or name:find("eject") or name:find("leave") or name:find("unhide") then
					if name:find("locker") or name:find("hide") or name:find("closet") then
						return nil -- Chặn lệnh tự thoát khỏi tủ
					end
				end
			end
			return oldNC(self, ...)
		end)
	end
end)

function GitAud(soundgit, filename) 
	local url = soundgit 
	local FileName = filename 
	writefile(FileName .. ".mp3", game:HttpGet(url)) 
	return (getcustomasset or getsynasset)(FileName .. ".mp3") 
end 

function CustomGitSound(soundlink, volume, filename, soundName, parent, looped, autoPlay) 
	local sound = Instance.new("Sound") 
	if tostring(soundlink):find("rbxassetid://") or tonumber(soundlink) then
		sound.SoundId = typeof(soundlink) == "number" and ("rbxassetid://" .. soundlink) or soundlink
	else
		sound.SoundId = GitAud(soundlink, filename) 
	end
	sound.Parent = parent or workspace 
	sound.Volume = volume or 1 
	sound.RollOffMaxDistance = 400 
	sound.RollOffMinDistance = 5 
	sound.PlaybackSpeed = 1 
	sound.Name = soundName or "bgtheme" 
	sound.Looped = looped or false 
	if autoPlay ~= false then
		sound:Play() 
	end
	return sound 
end 

-- Hàm Tween CFrame Model mượt mà
local function TweenModelCFrame(model, targetCFrame, duration)
	if not model then return end
	local cfVal = Instance.new("CFrameValue")
	cfVal.Value = model:GetPivot()
	local conn = cfVal:GetPropertyChangedSignal("Value"):Connect(function()
		if model and model.Parent then
			model:PivotTo(cfVal.Value)
		end
	end)
	local tween = TweenService:Create(cfVal, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {Value = targetCFrame})
	tween:Play()
	tween.Completed:Connect(function()
		conn:Disconnect()
		cfVal:Destroy()
	end)
end

-- HỆ THỐNG CAMERA SHAKER TỐI ƯU HÓA MEMORY
local function initializeBypassedShaker() 
	local playerCamera = workspace.CurrentCamera 
	if not playerCamera then return nil end 
	
	local function cameraCallback(shakeCFrame) 
		playerCamera.CFrame = initialCamCFrame * shakeCFrame 
	end 
	
	local renderPriority = Enum.RenderPriority.Camera.Value + 1 
	local shakerInstance = { 
		["_running"] = false, 
		["_renderName"] = "BypassedCameraShaker", 
		["_renderPriority"] = renderPriority, 
		["_camShakeInstances"] = {}, 
		["_removeInstances"] = {}, 
		["_callback"] = cameraCallback 
	} 
	RunService:BindToRenderStep(shakerInstance._renderName, shakerInstance._renderPriority, function(deltaTime) 
		local posAccumulator = Vector3.zero
		local rotAccumulator = Vector3.zero 
		local instances = shakerInstance._camShakeInstances 
		local removeCount = 0
		
		for i = 1, #instances do 
			local inst = instances[i] 
			if inst.currentFadeTime <= 0 and not inst.sustain then 
				removeCount += 1
				shakerInstance._removeInstances[removeCount] = i 
			else 
				local deltaTick = inst.tick 
				local fadeTime = inst.currentFadeTime 
				local noiseVal = Vector3.new(math.noise(deltaTick, 0) * 0.5, math.noise(0, deltaTick) * 0.5, math.noise(deltaTick, deltaTick) * 0.5) 
				if inst.fadeInDuration > 0 and inst.sustain then 
					if fadeTime < 1 then fadeTime = fadeTime + deltaTime / inst.fadeInDuration end 
				end 
				if not inst.sustain then fadeTime = fadeTime - deltaTime / inst.fadeOutDuration end 
				if inst.sustain then 
					inst.tick = deltaTick + deltaTime * inst.Roughness * inst.roughMod 
				else 
					inst.tick = deltaTick + deltaTime * inst.Roughness * inst.roughMod * fadeTime 
				end 
				inst.currentFadeTime = math.clamp(fadeTime, 0, 1) 
				local shakeVector = noiseVal * inst.Magnitude * inst.magnMod * fadeTime 
				posAccumulator = posAccumulator + shakeVector * inst.PositionInfluence 
				rotAccumulator = rotAccumulator + shakeVector * inst.RotationInfluence 
			end 
		end 
		for i = removeCount, 1, -1 do 
			table.remove(instances, shakerInstance._removeInstances[i]) 
			shakerInstance._removeInstances[i] = nil 
		end 
		local finalCFrame = CFrame.new(posAccumulator) * CFrame.Angles(0, math.rad(rotAccumulator.Y), 0) * CFrame.Angles(math.rad(rotAccumulator.X), 0, math.rad(rotAccumulator.Z)) 
		shakerInstance._callback(finalCFrame) 
	end) 
	shakerInstance._running = true 
	return shakerInstance 
end

		local y = math.clamp(mousePos.Y - parentPos.Y - circleSize.Y/2, -Overflow, parentSize.Y - circleSize.Y + Overflow) 
		targetPosition = UDim2.fromOffset(x, y) 
	end)

-- HÀM ĐẬP TỦ (HẤT VĂNG)
local function ExecutePush()
	if Frozen or Ended then return end
	IsBeingPushed = true 
	CurrentFollowSpeed = PushFollowSpeed 

	local parent = playerCircle.Parent 
	local parentSize = parent.AbsoluteSize 
	local circleSize = playerCircle.AbsoluteSize 

	local angles = {0, math.pi/2, math.pi, math.pi*1.5, math.pi/4, math.pi*0.75}
	local angle = angles[math.random(#angles)]
	local distance = math.max(parentSize.X, parentSize.Y) * 0.85

	local currentPos = Vector2.new(playerCircle.Position.X.Offset, playerCircle.Position.Y.Offset)
	local pushOffsetVec = Vector2.new(math.cos(angle), math.sin(angle)) * distance
	local targetVec = currentPos + pushOffsetVec

	local Overflow = 90
	local clampX = math.clamp(targetVec.X, -Overflow, parentSize.X - circleSize.X + Overflow)
	local clampY = math.clamp(targetVec.Y, -Overflow, parentSize.Y - circleSize.Y + Overflow)

	targetPosition = UDim2.fromOffset(clampX, clampY)
	smoothPosition = targetPosition
	PushForce = pushOffsetVec.Unit * Config.Gameplay.PushForce

	PlayRandomPushSound()
	PlayShockWave()

	task.delay(Config.Gameplay.PushLockDuration, function()
		IsBeingPushed = false
		CurrentFollowSpeed = PlayerFollowSpeed
	end)
end

		ActiveShaker._camShakeInstances = {}

-- Render loop cập nhật vị trí Circle (Tối ưu nhẹ nhàng)
RunService.RenderStepped:Connect(function(dt) 
	if Frozen or Ended then return end 

	shakeTime += dt 

	if not IsBeingPushed then
		MoveVelocity *= MoveFriction 
		targetPosition = targetPosition + UDim2.fromOffset(MoveVelocity.X * dt, MoveVelocity.Y * dt) 
	end

	local parent = playerCircle.Parent 
	if not parent then return end
	local parentSize = parent.AbsoluteSize 
	local circleSize = playerCircle.AbsoluteSize 
	local Overflow = 90 
	targetPosition = UDim2.fromOffset( 
		math.clamp(targetPosition.X.Offset, -Overflow, parentSize.X - circleSize.X + Overflow), 
		math.clamp(targetPosition.Y.Offset, -Overflow, parentSize.Y - circleSize.Y + Overflow) 
	) 

	local alpha = math.clamp(dt * CurrentFollowSpeed, 0, 1) 
	local pushOffset = UDim2.fromOffset(PushForce.X * dt, PushForce.Y * dt) 

	smoothPosition = smoothPosition:Lerp(targetPosition + pushOffset, alpha) 
	PushForce *= 0.85 

	local ox = math.sin(shakeTime * shakeSpeedX) * shakeAmount + math.cos(shakeTime * (shakeSpeedX * 1.6)) * (shakeAmount * 0.35) 
	local oy = math.cos(shakeTime * shakeSpeedY) * shakeAmount + math.sin(shakeTime * (shakeSpeedY * 1.6)) * (shakeAmount * 0.35) 

	playerCircle.Position = smoothPosition + UDim2.fromOffset(ox, oy) 
end) 

-- THUẬT TOÁN 4 GÓC MÀN HÌNH + KIỂM TRA KHOẢNG CÁCH TỐI THIỂU
local function GetFarApartPositions(container)
	local containerSize = container.AbsoluteSize
	local width = containerSize.X > 100 and containerSize.X or 800
	local height = containerSize.Y > 100 and containerSize.Y or 600
	
	local padding = 50
	local circleDiameter = 90
	local halfW = (width - padding * 2) / 2
	local halfH = (height - padding * 2) / 2
	
	local quadrants = {
		{ minX = padding, maxX = padding + halfW - circleDiameter, minY = padding, maxY = padding + halfH - circleDiameter },
		{ minX = padding + halfW, maxX = width - padding - circleDiameter, minY = padding, maxY = padding + halfH - circleDiameter },
		{ minX = padding, maxX = padding + halfW - circleDiameter, minY = padding + halfH, maxY = height - padding - circleDiameter },
		{ minX = padding + halfW, maxX = width - padding - circleDiameter, minY = padding + halfH, maxY = height - padding - circleDiameter },
	}
	
	for i = #quadrants, 2, -1 do
		local j = math.random(i)
		quadrants[i], quadrants[j] = quadrants[j], quadrants[i]
	end
	
	local minDistance = math.max(220, math.min(width, height) * 0.35)
	
	local positions = {}
	for i = 1, 3 do
		local quad = quadrants[i]
		local selectedPos = nil
		local attempts = 0
		
		repeat
			attempts += 1
			local rx = math.random(math.floor(quad.minX), math.floor(math.max(quad.minX + 10, quad.maxX)))
			local ry = math.random(math.floor(quad.minY), math.floor(math.max(quad.minY + 10, quad.maxY)))
			local candidateVec = Vector2.new(rx, ry)
			
			local isFarEnough = true
			for _, pos in ipairs(positions) do
				local prevVec = Vector2.new(pos.X.Offset, pos.Y.Offset)
				if (candidateVec - prevVec).Magnitude < minDistance then
					isFarEnough = false
					break
				end
			end
			
			if isFarEnough or attempts > 25 then
				selectedPos = UDim2.fromOffset(rx, ry)
			end
		until selectedPos
		
		table.insert(positions, selectedPos)
	end
	
	return positions
end

			for _, targetData in ipairs(activeTargets) do
				if not targetData.IsCompleted then
					local tCenter = targetData.Frame.AbsolutePosition + targetData.Frame.AbsoluteSize / 2
					local tRadius = math.min(targetData.Frame.AbsoluteSize.X, targetData.Frame.AbsoluteSize.Y) / 2
					local isHovering = (playerCenter - tCenter).Magnitude <= tRadius
					
					if isHovering then
						targetData.Progress = math.clamp(targetData.Progress + (dt / Config.Gameplay.CircleFillTime), 0, 1)
						targetData.BarFill.Size = UDim2.new(targetData.Progress, 0, 1, 0)
						
						if targetData.Progress >= 1 then
							targetData.IsCompleted = true
							completedCount += 1
							
							remainingTime = math.min(phaseDuration, remainingTime + Config.Gameplay.TimeBonusOnComplete)
							
							local frameToDestroy = targetData.Frame
							task.spawn(function()
								for _, desc in ipairs(frameToDestroy:GetDescendants()) do
									if desc:IsA("ImageLabel") then
										TweenService:Create(desc, TweenInfo.new(Config.Gameplay.FadeOutDuration), {ImageTransparency = 1}):Play()
									elseif desc:IsA("Frame") then
										TweenService:Create(desc, TweenInfo.new(Config.Gameplay.FadeOutDuration), {BackgroundTransparency = 1}):Play()
									end
								end
								if frameToDestroy:IsA("ImageLabel") then
									TweenService:Create(frameToDestroy, TweenInfo.new(Config.Gameplay.FadeOutDuration), {ImageTransparency = 1}):Play()
								end
								task.wait(Config.Gameplay.FadeOutDuration)
								if frameToDestroy and frameToDestroy.Parent then
									frameToDestroy:Destroy()
								end
							end)
						end
					end
				end
			end

			if targetData.Frame and targetData.Frame.Parent then

			if next(goals) then 
				local t = TweenService:Create(v, TweenInfo.new(2, Enum.EasingStyle.Quad), goals)
